"""確定性前置掃描（recheck_scan）：掃 定稿/*.jsonl，貼 risk_flags，只輸出報告、不改資料。

規格（reaudit.txt 第 1/3 篇 §三、第 3/3 篇 §二）：

 10. 雙欄總數詞／無標點多「云言問答曰」／破折號長插注／序跋高風險
 11. FALSE_FRIEND_HIT（30+ 高危假朋友）＋ BOOK_TERM_HIT（各書具體詞旗標）

本檔只讀取 定稿/*.jsonl；不寫任何 定稿／docs／材料檔。
"""
import argparse
import datetime
import io
import json
import os
import re
import sys

B = r'C:\Users\lionc\OneDrive\桌面\XW\三大部'

BOOKS = {
    '業疏': 'yeshu',
    '戒疏': 'jieshu',
    '尼鈔': 'nicao',
    '行事鈔': 'xingshichao',
    '義鈔': 'yichao',
    '資持記': 'zhichi',
}

LABELS = ('疏文', '記文', '鈔文', '律文', '註文', '序文', '記序')
LEAD_LAB = re.compile(r'^\s*【(%s)】' % '|'.join(LABELS))
SRC_HEAD_BAD = re.compile(r'^[】」』、，,。；;：:）)\]】]')
SRC_TAIL_OK = '。！？…」』）〕】〉》”'
BAD_VAR = ('却', '鉢', '厨', '徧', '牀', '説', '衆')
BG_TAGS = ('【前文背景', '【後文背景', '【所釋鈔文', '【所釋疏文', '【待審譯文',
           '【本次待審', '【裁決檢索摘錄', '【原典夾注', '【原典雙行夾注',
           '【資持記相關釋文', '【義鈔輔要記', '【分段核驗表】', '【必改硬傷】',
           '【逐句簡核】', '〔定稿殘留')
TP_WORDS = ('道宣', '元照', '南山律師', '南山', '律祖')

PUNCT = '，。；：！？、'
TOTAL_WORDS = ('八事', '五事', '四門', '十門', '二者', '兩者', '九事', '六事', '三事', '七事')
PREFACE_RE = re.compile(r'^\s*(?:序|敘|叙|原夫|蓋聞|竊以|謹序|題辭)|卷首|會本|江戶')

# ---- 高危假朋友（全書通用；只作 0 成本旗標，不進提示詞） ----
FALSE_FRIEND = (
    '中國', '人語', '通解', '餘論', '入宗', '一期', '餘食法', '八遮', '通局',
    '下手疏', '口說', '投竄', '眼暗', '來者', '何法', '長者', '須用物', '惡捉',
    '上中下', '背別', '供須', '已將久', '為妨', '制意', '中氣', '二者',
    '疏人', '造逆', '教通', '克定', '有通有局',
)
# ---- 各書抽樣已證實的具體詞（旗標用；提示詞不得引用） ----
BOOK_TERMS = {
    '尼鈔': ('刮刀', '波羅夷', '房舍田園', '招提', '十三步', '體毛', '髮長', '僧物', '式叉摩那'),
    '行事鈔': ('不作餘食法', '八遮', '有通有局', '下手疏', '何法持三衣', '投竄', '教通餘論', '克定一期'),
    '義鈔': ('執情三品', '背別', '目連', '疏人', '三家', '上中下', '婆沙論', '本自無過'),
    '業疏': ('八事', '九事', '得在浮沈', '隱顯互見', '初文隨相', '非受體', '今已將久'),
    '戒疏': ('六七百年', '江戶', '如上所引', '罪相等故', '中氣', '二者', '記序'),
    '資持記': ('所釋鈔文', '某下', '此下', '牒字', '一期'),
}

# 致命（資料層確定壞掉；缺開頭標籤屬各書慣例問題，列風險旗標不列致命）
FATAL = {'EMPTY_TR', 'MISSING_FIELD', 'DUAL_MISSING_COL', 'SRC_HEAD_CLOSE',
         'SRC_TAIL_NO_END', 'BG_LEAK', 'BAD_VAR_QUOTE',
         'BRACKET_OVERFLOW', 'TAIL_COLON', 'DE_DENG', 'Q_WENDENG'}
FLAG_ORDER = (
    'EMPTY_TR', 'MISSING_FIELD', 'DUAL_MISSING_COL', 'SRC_HEAD_CLOSE',
    'SRC_TAIL_NO_END', 'NO_LEAD_LABEL', 'BG_LEAK', 'GAP_SEG', 'THIRD_PERSON',
    'BAD_VAR_QUOTE', 'BRACKET_OVERFLOW', 'TAIL_COLON', 'DE_DENG', 'Q_WENDENG',
    'INFLATE', 'SHRINK', 'LONE_CORNER', 'DUAL_TOTAL', 'NO_PUNCT_DIALOG',
    'DASH_NOTE', 'PREFACE_RISK', 'FALSE_FRIEND_HIT', 'BOOK_TERM_HIT',
)
_HAN = re.compile(r'[\u4e00-\u9fff]')


def han(s):
    return ''.join(_HAN.findall(s or ''))


def _parts(row):
    segs = [s for s in (row.get('segments') or [])
            if (s.get('source') or s.get('translation'))]
    if segs:
        return [{'lab': s.get('label') or row.get('label') or '',
                 'orig': s.get('source') or '',
                 'tr': s.get('translation') or ''} for s in segs]
    return [{'lab': row.get('label') or '',
             'orig': row.get('source') or '',
             'tr': row.get('translation') or ''}]


def _ratio_plain(tr):
    """漢字數（排除夾注／校勘括號內容，避免夾注白話重複計入膨脹率）。"""
    t = re.sub(r'（[^）]*）|\([^)]*\)|〔[^〕]*〕', '', tr or '')
    return len(han(t))


def _quote_bad(tr, src):
    """白話引號內殘留 BAD_VAR；src 同字者視為被訓釋原詞，豁免。"""
    for q in re.findall(r'[「『]([^」』]{1,})[」』]', tr or ''):
        for ch in BAD_VAR:
            if ch in q and ch not in (src or ''):
                return ch
    return ''


def scan_row(book, row, raw_h=None, inflate_thr=1.9, shrink_thr=0.5):
    """回傳 {'flags': [...], 'detail': {...}}。inflate_thr 由 scan_book 依全書中位數調整。"""
    flags, detail = [], {}

    def add(f, **kw):
        if f not in flags:
            flags.append(f)
        if kw:
            detail.setdefault(f, {}).update(kw)

    key = str(row.get('key') or '')
    tr = row.get('translation')
    src = row.get('source') or ''
    parts = _parts(row)

    if 'translation' not in row or tr is None:
        add('MISSING_FIELD', field='translation')
        tr = ''
    if not isinstance(tr, str):
        add('MISSING_FIELD', field='translation(type)')
        tr = ''
    if not src:
        add('MISSING_FIELD', field='source')

    sh = len(han(src))
    oh = _ratio_plain(tr)
    if not han(tr):
        add('EMPTY_TR', han=0)
    _by_lab = {}
    for p in parts:
        if p['lab']:
            _by_lab.setdefault(p['lab'], []).append(p)
    if len(_by_lab) > 1:
        _empty = [lab for lab, ps in _by_lab.items()
                  if ps and all(not han(x['tr']) for x in ps)]
        if _empty:
            add('DUAL_MISSING_COL', empty=_empty)

    _s = src.strip()
    if _s and SRC_HEAD_BAD.match(_s):
        add('SRC_HEAD_CLOSE', head=_s[:12])
    # 短科判標目（無句讀）不強求句末標點，避免整本書的科名洗版
    if _s and _s[-1] not in SRC_TAIL_OK and (sh >= 20 or any(c in _s for c in '。，；')):
        add('SRC_TAIL_NO_END', tail=_s[-12:])

    if tr and not LEAD_LAB.match(tr):
        add('NO_LEAD_LABEL', head=tr.strip()[:16])
    leaked = [t for t in BG_TAGS if t in (tr or '')]
    if leaked:
        add('BG_LEAK', tags=leaked)

    if GAP_RE.search(key):
        add('GAP_SEG', key=key)

    if tr and any(w in tr for w in TP_WORDS) and not any(w in src for w in ('道宣', '元照', '南山', '律祖')):
        add('THIRD_PERSON', words=[w for w in TP_WORDS if w in tr][:3])

    _bv = _quote_bad(tr, src)
    if _bv:
        add('BAD_VAR_QUOTE', char=_bv)

    if (tr or '').count('〔') > src.count('〔'):
        add('BRACKET_OVERFLOW', out=tr.count('〔'), src=src.count('〔'))

    _t = (tr or '').rstrip()
    if _t.endswith(('：', ':')):
        add('TAIL_COLON', tail=_t[-10:])
    if '的等。' in (tr or ''):
        add('DE_DENG')
    if re.search(r'[？?]等等。', tr or ''):
        add('Q_WENDENG')

    if sh >= 10 and oh > sh * inflate_thr:
        add('INFLATE', src_han=sh, out_han=oh, ratio=round(oh / sh, 2),
            threshold=inflate_thr)
    if sh >= 10 and oh < sh * shrink_thr:
        add('SHRINK', src_han=sh, out_han=oh, ratio=round((oh / sh) if sh else 0, 2),
            threshold=shrink_thr)
    if tr and '「' not in tr and ('『' in tr or '』' in tr):
        add('LONE_CORNER')

    if len(parts) > 1:
        _hit = [w for w in TOTAL_WORDS if w in tr or w in src]
        if _hit:
            add('DUAL_TOTAL', words=_hit[:4])

    _punct = sum(src.count(ch) for ch in PUNCT)
    if sh >= 60 and _punct <= 1 and len(DIALOG_RE.findall(src)) >= 2:
        add('NO_PUNCT_DIALOG', punct=_punct, dialog=len(DIALOG_RE.findall(src)))

    if '——' in (tr or ''):
        add('DASH_NOTE')
    if '序文' == (row.get('label') or '') or PREFACE_RE.search(key + ' ' + src[:60]):
        add('PREFACE_RISK')

    ff = [w for w in FALSE_FRIEND if w in src]
    if ff:
        add('FALSE_FRIEND_HIT', words=ff[:8])
    bt = [w for w in BOOK_TERMS.get(book, ()) if w in src]
    if bt:
        add('BOOK_TERM_HIT', words=bt[:8])

    return {'book': book, 'key': key,
            'vol': row.get('vol') or '', 'chunk': row.get('chunk') or '',
            'src_han': sh, 'tr_han': oh, 'flags': flags, 'detail': detail}


def scan_book(book, jsonl=None):
    bid = jsonl or BOOKS[book]
    p = os.path.join(B, '定稿', bid + '.jsonl')
    if not os.path.exists(p):
        raise SystemExit('找不到定稿：%s' % p)
    raws, bad = [], []
    with io.open(p, encoding='utf-8') as f:
        for line in f:
            if not line.strip():
                continue
            try:
                raws.append(json.loads(line))
            except Exception as e:
                bad.append({'book': book, 'key': '?', 'vol': '', 'chunk': '',
                            'src_han': 0, 'tr_han': 0,
                            'flags': ['MISSING_FIELD'],
                            'detail': {'MISSING_FIELD': {'error': str(e)[:80]}}})
    # 各書底本長度／譯文長度比例差異大（業疏中位數約 2.1），
    # 絕對 1.9 會把正常段全打成膨脹；以「全書中位數 ×1.25，且不低於 1.9」為門檻。
    _rs = []
    for r in raws:
        sh = len(han(r.get('source')))
        if sh >= 10:
            _rs.append(_ratio_plain(r.get('translation')) / sh)
    _rs.sort()
    median = _rs[len(_rs) // 2] if _rs else 0.0
    inflate_thr = max(1.9, round(median * 1.25, 2)) if median else 1.9
    rows = [scan_row(book, r, inflate_thr=inflate_thr) for r in raws]
    rows.extend(bad)
    return {'book': book, 'file': '定稿/%s.jsonl' % bid, 'rows': len(rows),
            'median_ratio': round(median, 2), 'inflate_thr': inflate_thr,
            'items': rows}


def summarize(result):
    items = result['items']
    counts = {}
    keys_by = {}
    n_fatal = 0
    for it in items:
        _fatal = [f for f in it['flags'] if f in FATAL]
        if _fatal:
            n_fatal += 1
        for f in it['flags']:
            counts[f] = counts.get(f, 0) + 1
            keys_by.setdefault(f, []).append(it['key'])
    return {'book': result['book'], 'file': result['file'], 'rows': result['rows'],
            'flagged': sum(1 for it in items if it['flags']),
            'fatal_rows': n_fatal,
            'median_ratio': result.get('median_ratio'),
            'inflate_thr': result.get('inflate_thr'),
            'counts': {f: counts[f] for f in FLAG_ORDER if f in counts},
            'keys': {f: keys_by[f] for f in FLAG_ORDER if f in keys_by}}


def _fmt_summary(s):
    head = '%-6s %6s 筆｜有旗標 %5s｜致命 %4s｜比例中位 %s（膨脹門檻 %s）' % (
        s['book'], s['rows'], s['flagged'], s['fatal_rows'],
        s.get('median_ratio'), s.get('inflate_thr'))
    bits = []
    for f, c in s['counts'].items():
        bits.append('%s %d' % (f, c))
    return head + ('\n        ' + '；'.join(bits) if bits else '')

    src, tr = '', ''
    if i0 > 0 and order[i0 - 1]['vol'] == u0['vol']:
        tr = _clip_tail(order[i0 - 1]['parts'][-1]['tr'], pre_tr)
        buf, acc = [], 0
        for j in range(i0 - 1, -1, -1):
            if order[j]['vol'] != u0['vol']:
                break
            t = '\n'.join(p['orig'] for p in order[j]['parts'])
            buf.append(t)
            acc += len(han(t))
            if acc >= pre_src:
                break
        src = _clip_tail('\n'.join(reversed(buf)), pre_src)
    return src, tr


def context_after(order, i1, post_src=None, post_tr=None):
    post_src = POST_SRC_CAP if post_src is None else post_src
    post_tr = POST_TR_CAP if post_tr is None else post_tr
    u1 = order[i1]
    src, tr = '', ''
    if i1 + 1 < len(order) and order[i1 + 1]['vol'] == u1['vol']:
        tr = _clip_head(order[i1 + 1]['parts'][0]['tr'], post_tr)
        buf, acc = [], 0
        for j in range(i1 + 1, len(order)):
            if order[j]['vol'] != u1['vol']:
                break
            t = '\n'.join(p['orig'] for p in order[j]['parts'])
            buf.append(t)
            acc += len(han(t))
            if acc >= post_src:
                break
        src = _clip_head('\n'.join(buf), post_src)
    return src, tr


def _gap_kind(order, i, raw_h):
    a = order[i - 1]['raw_end']
    b = order[i]['raw_start']
    if raw_h is None or not (0 <= a < b <= len(raw_h)):
        return '正文'
    seg = raw_h[a:b]
    for hint in NONBODY_HINTS:
        if hint in seg:
            return '非正文'
    return '正文'


def _intra_gaps(unit):
    out = []
    parts = unit['parts']
    for p, q in zip(parts, parts[1:]):
        if 0 <= p.get('end', -1) < q.get('start', -1):
            n = q['start'] - p['end']
            if n >= GAP_MIN:
                out.append(n)
    return out


def unit_size(u, mode):
    src = sum(len(p['orig']) for p in u['parts'])
    if mode == 'source':
        return src
    return src + sum(len(p['tr']) for p in u['parts'])

def _interlinear_notes(unit, cap=900):
    """原典雙行夾注：取 display_source 的〔…〕（CBETA 尼鈔 X40n0724 還原）。"""
    notes = []
    for p in unit['parts']:
        disp = p.get('display') or ''
        for m in re.finditer(r'〔([^〕]{2,})〕', disp):
            a = max(0, m.start() - 10)
            notes.append('%s〔%s〕' % (disp[a:m.start()].strip(), m.group(1).strip()))
    return '\n'.join(notes[:10])[:cap]


def _plain_src(unit):
    return '\n'.join(p['orig'] for p in unit['parts'] if (p.get('orig') or '').strip())


def _parallel_notes(book, src, cap=900, limit=2):
    """書庫平行釋文：優先同宗註疏（戒疏／行宗記／資持記／行事鈔／婆沙）。"""
    try:
        import local_corpus as LC
    except Exception:
        return ''
    q = han(src)[:14]
    if len(q) < 6:
        return ''
    try:
        hits = LC.find(q, None, limit=limit, pad=150,
                       ctx_book=('義鈔' if book == '義鈔' else book))
    except Exception:
        return ''
    out = []
    for h in hits or []:
        b = h.get('book') or ''
        if b == book or b in ('義鈔輔要記',):
            continue
        kw = (h.get('kwic') or '').strip()
        if kw:
            out.append('《%s》%s：%s' % (b, h.get('juan') or '', kw))
    return '\n'.join(out)[:cap]


# 義鈔平行釋文用的律學主題詞（只作檢索鍵，不進提示詞）
YICHAO_TOPICS = ('三衣', '說淨', '坐具', '臥具', '波羅夷', '僧殘', '捨墮', '目連',
                 '耆婆', '黃門', '二形', '上中下', '戒體', '無作', '覆藏', '懺',
                 '羯磨', '二篇', '眾學', '滅擯')


def _yichao_parallel(src, cap=900, limit=2):
    """義鈔：以主題詞檢索戒疏／行宗記／資持記／行事鈔的平行釋文。"""
    try:
        import local_corpus as LC
    except Exception:
        return ''
    out = []
    for term in YICHAO_TOPICS:
        if term not in src:
            continue
        for work in ('戒疏', '行宗記', '資持記', '行事鈔'):
            try:
                hs = LC.find(term, work, limit=1, pad=110)
            except Exception:
                hs = []
            if hs:
                h = hs[0]
                out.append('《%s》%s（因「%s」檢得）：%s'
                           % (h.get('book'), h.get('juan') or '', term,
                              (h.get('kwic') or '').strip()))
                break
        if len(out) >= limit:
            break
    return '\n'.join(out)[:cap]


def _ke_note(book, src, cap=360):
    """科判位置：業疏科／戒疏科取最相近一筆。"""
    try:
        import local_corpus as LC
        work = '業疏科' if book == '業疏' else '戒疏科'
        q = han(src)[:10]
        if len(q) < 5:
            return ''
        hits = LC.find(q, work, limit=1, pad=80)
        if hits:
            return ('《%s》%s：%s' % (hits[0].get('book'), hits[0].get('juan') or '',
                                     hits[0].get('kwic') or '')).strip()[:cap]
    except Exception:
        return ''
    return ''


def _suoshu_text(order, i, cap=1200):
    """單邊記文（含 JE-/YE-/GAP/BLIND 殘片）：往前找同一卷最近的【疏文】，作為【所釋疏文】。

    已含【疏文】的雙欄單位不重複掛（那屬雙向對賬，不是單邊記文）。
    """
    u = order[i]
    _is_ji = any(p['lab'] == '記文' and (p.get('orig') or '').strip()
                 for p in u['parts'])
    _has_su = any(p['lab'] == '疏文' and (p.get('orig') or '').strip()
                  for p in u['parts'])
    _gap = bool(re.match(r'(?:JE|YE|GAP|BLIND)', u['key'] or ''))
    # 單邊記文＝只有記文沒有疏文；缺口殘片（JE/YE/GAP/BLIND）即使是單欄也掛，
    # 但已同時含疏＋記的雙欄單位不重複掛。
    if not ((_is_ji and not _has_su) or (_gap and not (_has_su and _is_ji))):
        return ''
    for j in range(i - 1, max(-1, i - 10), -1):
        v = order[j]
        if u['vol'] and v['vol'] and v['vol'] != u['vol']:
            break
        if v['raw_start'] >= u['raw_start'] and u['raw_start'] > 0:
            continue
        if u['raw_start'] > 0 and v['raw_end'] >= 0 \
                and u['raw_start'] - v['raw_end'] > 6000:
            continue
        sus = [p['orig'] for p in v['parts']
               if p['lab'] == '疏文' and (p.get('orig') or '').strip()]
        if sus:
            return _clip_tail('\n'.join(sus), cap)
    return ''


def reference_blocks(book, unit, order, i):
    """回傳 [(標題, 內文)]：只掛本段可用且有據的對照層。"""
    blocks = []
    src = _plain_src(unit)
    if book == '尼鈔':
        notes = _interlinear_notes(unit)
        if notes:
            blocks.append((
                '原典雙行夾注（CBETA X40n0724；〔…〕內為夾注）',
                '下列夾注皆為原典所有，譯文以圓括號（）對譯者必須保留，'
                '不得當作「自加解說括號」刪除；亦不得把夾注內容改寫成正文。\n' + notes))
    if book == '行事鈔':
        try:
            import mother_corpus as MC
            if not getattr(MC, '_RA_SEQ_READY', False):
                _seq = MC._seq_json()
                MC._seq_json = (lambda: _seq)
                MC._RA_SEQ_READY = True
            ji = MC.ji_for_chao(src, cap=900)
        except Exception:
            ji = ''
        if ji:
            blocks.append(('資持記相關釋文（僅供訓釋字義，嚴禁搬入正文）', ji))
    if book == '義鈔':
        disp = '\n'.join(p.get('display') or '' for p in unit['parts']
                         if (p.get('display') or '').strip())
        if not disp:
            disp = (unit.get('row') or {}).get('display_source') or ''
        if disp and han(disp):
            blocks.append(('原典補標點原文（僅供斷句對照）', disp[:REF_CAP]))
        par = _yichao_parallel(src)
        if par:
            blocks.append(('平行釋文（戒疏／行宗記／資持記／行事鈔；僅供訓釋，嚴禁搬入正文）', par))
    if book in ('業疏', '戒疏'):
        sus = _suoshu_text(order, i)
        if sus:
            blocks.append(('所釋疏文（單邊記文之母文，僅供對讀，嚴禁抄入記文）', sus))
        ke = _ke_note(book, src)
        if ke:
            blocks.append(('所屬科判位置（僅供定位，勿改寫正文）', ke))
        if book == '戒疏' and ('文云' in src or '注云' in src):
            try:
                import local_corpus as LC
                q = han(src)[:12]
                hits = LC.find(q, '含注戒本', limit=1, pad=120)
                if hits:
                    blocks.append(('《含注戒本》對照（文云／注云出處）',
                                   ('%s：%s' % (hits[0].get('juan') or '',
                                                hits[0].get('kwic') or '')).strip()))
            except Exception:
                pass
    return blocks


def render_material(book, pack, idx, book_id, key, pre, post, gaps, mode, raws,
                    mother='', refs=None):
    vol = pack[0]['vol'] or '（未分卷）'
    n = len(pack)
    src_chars = sum(len(p['orig']) for u in pack for p in u['parts'])
    tr_chars = sum(len(p['tr']) for u in pack for p in u['parts'])
    labs = sorted({p['lab'] for u in pack for p in u['parts']})
    src_all = '\n'.join(p['orig'] for u in pack for p in u['parts'])
    tr_all = '\n'.join(p['tr'] for u in pack for p in u['parts'])
    long_ = any(re.search(r'_L\d{3,}', u['key']) for u in pack) \
        or '（本段前文）' in src_all or '（本段後文）' in src_all
    notes = BC.notes_for(book, labs, src_all, tr_all, long=long_,
                         key=pack[0]['key'])
    prompt = BC.render(BC.REAUDIT_PACK_PROMPT_TMPL,
                       scope_note=(SINGLE_SCOPE if n == 1 else BC.PACK_SCOPE),
                       **notes)
    L = [prompt, '', '---', '',
         '（出處：《%s》%s，%s｜單位 %d–%d｜原文 %s 字＋現譯 %s 字）'
         % (book, vol, key, 1, n, _fmt(src_chars), _fmt(tr_chars)), '']
    _gaps_body = [g for g in gaps if g['kind'] == '正文']
    if _gaps_body:
        L += ['【正文缺口提示】', '',
              '（本檔前／後／段間有正文缺口共 %s 字：%s；此係定稿未收錄之正文，'
              '禁止代譯、補譯或憑上下文虛構，亦不得據此判定現譯漏譯。）'
              % (_fmt(sum(g['chars'] for g in _gaps_body)),
                 '、'.join('%s→%s %d字' % (g['after'], g['before'], g['chars'])
                          for g in _gaps_body)), '']
    L += ['【前文背景（僅供參考，嚴禁譯入正文）】', '']
    if pre[0].strip():
        L += ['■ 原文（前接）', '', pre[0], '']
    if pre[1].strip():
        L += ['■ 前鄰現譯', '', pre[1], '']
    if not pre[0].strip() and not pre[1].strip():
        L += ['（本卷開頭，無前文背景）', '']
    if refs:
        L += ['【對照層（僅供校勘，嚴禁譯入正文）】', '']
        for title, body in refs:
            if not (body or '').strip():
                continue
            L += ['■ %s' % title, '', body.strip(), '']
    _dirty = [(i, u) for i, u in enumerate(pack, 1) if u.get('dirty')]
    if _dirty:
        L += ['【定稿殘留提示】', '',
              '下列單位的原文／現譯含材料區塊殘留標記（逐字照錄，未刪改）：%s。'
              '請一併判定現譯是否誤將背景或清單混入正文，並於【必改硬傷】點名。'
              % '、'.join('[%02d]（%s）' % (i, '／'.join(u['dirty'])) for i, u in _dirty), '']
    L += ['【本次待審單元】', '']
    for i, u in enumerate(pack, 1):
        tag = '[%02d]' % i
        L += ['### %s（key: %s）' % (tag, u['key']), '']
        for p in u['parts']:
            L += ['■ %s（原文）%s' % (p['lab'], tag), '']
            if p.get('dirty_orig'):
                L += ['〔定稿殘留：此段含 %s；逐字照錄〕' % '／'.join(p['dirty_orig']), '']
            L += [p['orig'].strip(), '']
            L += ['■ %s（現譯）%s' % (p['lab'], tag), '']
            if p.get('dirty_tr'):
                L += ['〔定稿殘留：此段含 %s；逐字照錄〕' % '／'.join(p['dirty_tr']), '']
            L += [p['tr'].strip(), '']
    L += ['【後文背景（僅供參考，嚴禁譯入正文）】', '']
    if post[0].strip():
        L += ['■ 原文（後接）', '', post[0], '']
    if post[1].strip():
        L += ['■ 後鄰現譯', '', post[1], '']
    if not post[0].strip() and not post[1].strip():
        L += ['（本卷末尾，無後文背景）', '']
    if mother:
        L += ['【所釋鈔文（僅供對照，嚴禁譯入正文）】', '', mother, '']
    return '\n'.join(L)


def _unit_json(idx, u):
    parts = []
    for p in u['parts']:
        parts.append({
            'lab': p['lab'],
            'orig': p['orig'],
            'tr': p['tr'],
            'src_len': len(p['orig']),
            'tr_len': len(p['tr']),
            'src_han': len(han(p['orig'])),
            'tr_han': len(han(p['tr'])),
            'start': p.get('start', -1),
            'end': p.get('end', -1),
            'raw_ok': p.get('raw_ok'),
            'dirty': list(p.get('dirty') or []),
            'dirty_orig': list(p.get('dirty_orig') or []),
            'dirty_tr': list(p.get('dirty_tr') or []),
            'src_hash': _h12(p['orig']),
            'hash': _h12('%s\t%s' % (p['orig'], p['tr'])),
        })
    return {
        'idx': idx,
        'key': u['key'],
        'chunk': u['chunk'],
        'vol': u['vol'],
        'raw_start': u['raw_start'],
        'raw_end': u['raw_end'],
        'labs': u['labs'],
        'parts': parts,
        'src_hash': u.get('src_hash') or _h12(
            han((u.get('row') or {}).get('source', '')) or
            ''.join(p['orig'] for p in u['parts'])),
        'hash': u['hash'],
        'raw_ok': u.get('raw_ok'),
        'parts_joined': bool(u.get('parts_joined', True)),
        'dirty': list(u.get('dirty') or []),
        'chars': unit_size(u, 'body'),
    }


def _render_len(book, pack, order, gap_mode, gap_min, raw_h, mode):
    """估算/實測材料長度；mode=md 時回傳整份 .md 字數（含資持記母本預留）。"""
    if mode != 'md':
        return sum(unit_size(u, mode) for u in pack)
    i0 = order.index(pack[0])
    i1 = order.index(pack[-1])
    pre = context_before(order, i0)
    post = context_after(order, i1)
    gaps = _pack_gaps(pack, order, gap_mode, gap_min, raw_h)
    txt = render_material(book, pack, 1, BOOKS[book]['id'],
                          'x' * (len(BOOKS[book]['id']) + 7), pre, post, gaps,
                          mode, raw_h, mother='')
    # 資持記母本在 emit 時才真正取用，這裡先預留上限，避免超長。
    return len(txt) + (MOTHER_CAP if book == '資持記' else 0)


# ---------------------------------------------------------------- 缺口收集
def _pack_gaps(pack, order, gap_mode, gap_min, raw_h):
    if gap_mode == 'off':
        return []
    gaps = []
    i0 = order.index(pack[0])
    i1 = order.index(pack[-1])
    if i0 > 0:
        prev = order[i0 - 1]
        n = pack[0]['raw_start'] - prev['raw_end']
        if prev['vol'] == pack[0]['vol'] and n >= gap_min and prev.get('mono') \
                and pack[0].get('mono'):
            gaps.append({'chars': n, 'after': prev['key'], 'before': pack[0]['key'],
                         'kind': _gap_kind(order, i0, raw_h), 'where': 'before'})
    if i1 + 1 < len(order):
        nxt = order[i1 + 1]
        n = nxt['raw_start'] - pack[-1]['raw_end']
        if nxt['vol'] == pack[-1]['vol'] and n >= gap_min and nxt.get('mono') \
                and pack[-1].get('mono'):
            gaps.append({'chars': n, 'after': pack[-1]['key'], 'before': nxt['key'],
                         'kind': _gap_kind(order, i1 + 1, raw_h), 'where': 'after'})
    for a, b in zip(pack, pack[1:]):
        n = b['raw_start'] - a['raw_end']
        if a['vol'] == b['vol'] and n >= gap_min and a.get('mono') and b.get('mono'):
            gaps.append({'chars': n, 'after': a['key'], 'before': b['key'],
                         'kind': _gap_kind(order, order.index(b), raw_h),
                         'where': 'inside'})
    for u in pack:
        for n in _intra_gaps(u):
            gaps.append({'chars': n, 'after': u['key'], 'before': u['key'],
                         'kind': '正文', 'where': 'inside'})
    return gaps


def build_packs(book, order, target, mode, gap_mode, gap_min, raw_h, group_by):
    packs, cur = [], []
    for i, u in enumerate(order):
        gap = None
        if cur:
            prev = cur[-1]
            same_vol = prev['vol'] == u['vol']
            n = u['raw_start'] - prev['raw_end'] if same_vol else 0
            if same_vol and n >= gap_min and prev.get('mono') and u.get('mono'):
                gap = {'chars': n, 'after': prev['key'], 'before': u['key'],
                       'kind': _gap_kind(order, i, raw_h), 'where': 'between'}
        boundary = bool(cur) and (
            cur[-1]['vol'] != u['vol']
            or (group_by == 'chunk' and cur[-1]['chunk'] != u['chunk']))
        _cut_gap = bool(gap) and gap['kind'] == '正文' and gap_mode == 'cut' \
            and key_tuple(u['key'])[0] == 0 and key_tuple(prev['key'])[0] == 0
        if cur and (boundary or _cut_gap):
            packs.append(cur)
            cur = []
        if cur:
            trial = cur + [u]
            if _render_len(book, trial, order, gap_mode, gap_min, raw_h,
                           mode) > target * (MD_TOL if mode == 'md' else 1.0):
                packs.append(cur)
                cur = []
        cur.append(u)
    if cur:
        packs.append(cur)
    return packs


# ---------------------------------------------------------------- 輸出
def emit_pack(outdir, pack, idx, book, book_id, mode, gaps, raws,
              pre=None, post=None, parent='', part='', key='', target=TARGET_CHARS,
              refs=None):
    key = key or '%s_r%04d' % (book_id, idx)
    pre = pre if pre is not None else ('', '')
    post = post if post is not None else ('', '')
    mother = mother_text(pack, book)
    md = render_material(book, pack, idx, book_id, key, pre, post, gaps, mode,
                         raws, mother, refs=refs)
    io.open(os.path.join(outdir, key + '.md'), 'w', encoding='utf-8').write(md)
    units_json = [_unit_json(i, u) for i, u in enumerate(pack, 1)]
    meta = {
        'key': key,
        'book': book,
        'book_id': book_id,
        'vol': pack[0]['vol'],
        'chunk': pack[0]['chunk'],
        'chunks': list(dict.fromkeys(u['chunk'] for u in pack)),
        'raw_start': min(u['raw_start'] for u in pack),
        'raw_end': max(u['raw_end'] for u in pack),
        'target_chars': target,
        'count_mode': mode,
        'chars': {
            'src': sum(len(p['orig']) for u in pack for p in u['parts']),
            'tr': sum(len(p['tr']) for u in pack for p in u['parts']),
            'total': sum(unit_size(u, 'body') for u in pack),
        },
        'md_chars': len(md),
        'md_bytes': len(md.encode('utf-8')),
        'gaps': gaps,
        'raw_verified': bool(raws),
        'dirty_units': [i for i, u in enumerate(pack, 1) if u.get('dirty')],
        'prev_unit': pack[0].get('prev_key', ''),
        'next_unit': pack[-1].get('next_key', ''),
        'pre_src': pre[0], 'pre_tr': pre[1],
        'post_src': post[0], 'post_tr': post[1],
        'mother': mother,
        'refs': [t for t, _b in (refs or [])],
        'parent': parent, 'part': part,
        'units': units_json,
        'hash': _h12('|'.join(u['hash'] for u in pack)),
    }
    io.open(os.path.join(outdir, key + '.json'), 'w', encoding='utf-8').write(
        json.dumps(meta, ensure_ascii=False, indent=1))
    return key, meta


def _safe_key(k):
    return re.sub(r'[^\w\u4e00-\u9fff.-]', '_', str(k))

def _write_raw_check(outdir, book, raws, order, bad_units, bad_parts,
                     disorder, raw_path):
    joined_bad = [u for u in order if u.get('parts_joined') is False]
    L = ['# 原文核驗（%s）' % book, '',
         '- 原文底本：%s' % (raw_path or '（無）'),
         '- 漢字流長度：%s' % ((_fmt(len(raws[0])) if raws and raws[0]
                             else '（無底本，raw 驗證 unavailable）')),
         '- 單位總數：%s' % _fmt(len(order)),
         '- source 與 raw 區間相符：%s' % _fmt(len(order) - len(bad_units)),
         '- source 與 raw 區間不符：%s' % _fmt(len(bad_units)),
         '- parts 拼接與 unit source 不一致：%s' % _fmt(len(joined_bad)),
         '- part 無法在 raw 內定位：%s' % _fmt(len(bad_parts)),
         '- 同卷 raw_start 非單調（定位可疑）：%s' % _fmt(len(disorder)),
         '']
    if bad_units:
        L += ['## source／raw 不符（前 50）', '',
              '| key | raw_start | raw_end | 定稿 source 起頭 | 底本同位起頭 |',
              '|:--|--:|--:|:--|:--|']
        for u in bad_units[:50]:
            a, b = u['raw_start'], u['raw_end']
            got = raws[0][a:b][:30] if raws and 0 <= a < b <= len(raws[0]) else ''
            L.append('| %s | %s | %s | %s | %s |'
                     % (u['key'], a, b, han(u['row'].get('source', ''))[:30], got))
        L.append('')
    if bad_parts:
        L += ['## parts 無法定位（前 50）', '',
              '| key | 段別 | part 原文起頭 |', '|:--|:--|:--|']
        for u, p in bad_parts[:50]:
            L.append('| %s | %s | %s |' % (u['key'], p['lab'], han(p['orig'])[:30]))
        L.append('')
    if joined_bad:
        L += ['## parts 拼接 ≠ unit source（前 50）', '',
              '| key | 段別 | 差異 |', '|:--|:--|--:|']
        for u in joined_bad[:50]:
            d = sum(len(han(p['orig'])) for p in u['parts']) - len(han(u['row'].get('source', '')))
            L.append('| %s | %s | %+d |' % (u['key'], '／'.join(u['labs']), d))
        L.append('')
    if disorder:
        L += ['## 同卷 raw_start 非單調（前 50）', '',
              '| key | vol | raw_start |', '|:--|:--|--:|']
        for u in disorder[:50]:
            L.append('| %s | %s | %s |' % (u['key'], u['vol'], u['raw_start']))
        L.append('')
    io.open(os.path.join(outdir, '_原文核驗.md'), 'w', encoding='utf-8').write(
        '\n'.join(L))

def generate(book, outdir, target, mode, gap_mode, gap_min, check_raw,
             limit=0, dry_run=False, force=False, group_by='vol', verbose=True):
    raws = load_raw(book)
    raw_h = raws[0]
    order = make_units(book, raw_h)
    for i, u in enumerate(order):
        u['prev_key'] = order[i - 1]['key'] if i > 0 else ''
        u['next_key'] = order[i + 1]['key'] if i + 1 < len(order) else ''
    disorder = [u for u in order if not u.get('mono')]
    gap_detect = (len(disorder) <= max(20, int(len(order) * 0.05)))
    effective_gap_mode = gap_mode if gap_detect else 'off'
    if gap_mode != 'off' and not gap_detect and verbose:
        print('  ⚠ %s raw_start 同卷順序可疑 %d／%d 筆（>5%%），'
              '依 gap 切檔會切出假缺口；本輪改用 gap-mode=off。'
              % (book, len(disorder), len(order)))
    packs = build_packs(book, order, target, mode, effective_gap_mode,
                        gap_min, raw_h, group_by)
    if limit:
        packs = packs[:limit]

        'book': book, 'book_id': BOOKS[book]['id'],
        'target_chars': target, 'count_mode': mode, 'group_by': group_by,
        'gap_mode': effective_gap_mode, 'gap_detect': gap_detect,
        'units': len(order), 'packs': len(packs),
        'chars': sum(unit_size(u, 'body') for u in order),
        'dirty': len([u for u in order if u.get('dirty')]),
        'raw': bool(raw_h), 'raw_path': raws[1],
    }
    if verbose:
        print('%s：定稿 %s 單位，打包 %d 檔（%s，約 %s 字%s）'
              % (book, _fmt(len(order)), len(packs),
                 '整份 md' if mode == 'md' else ('原文＋現譯' if mode == 'body' else '僅原文'),
                 _fmt(target), '；不跨 %s' % group_by))
    if dry_run:
        _sizes = [sum(unit_size(u, 'body') for u in p) for p in packs]
        _md = None
        if mode == 'md':

                      % book)
            else:
                bad_units = [u for u in order if u.get('raw_ok') is False]
                bad_parts = [(u, p) for u in order for p in u['parts']
                             if p.get('raw_ok') is False]
                joined_bad = [u for u in order if u.get('parts_joined') is False]
                print('  原文核驗：底本 %s；source 不符 %d；parts 拼接不符 %d；'
                      'part 定位不符 %d；順序可疑 %d'
                      % (raws[1] or '（無）', len(bad_units), len(joined_bad),
                         len(bad_parts), len(disorder)))

    os.makedirs(outdir, exist_ok=True)
    rep = os.path.join(outdir, '回覆')

            and not force:

                         '  確定重跑請加 --force，或改用新目錄。' % outdir)

                if re.match(r'.*_r\d{4}\.(md|json)$', f)]

        raise SystemExit('（已中止）輸出目錄已有再審材料（%d 個），'

            try:
                os.remove(os.path.join(outdir, f))
            except OSError:
                pass
    meta_all = []

        i0 = order.index(pack[0])

        else:
            joined_bad = [u for u in order if u.get('parts_joined') is False]

                  '順序可疑 %d → _原文核驗.md'

    print('已產生 %d 檔 → %s' % (len(meta_all), os.path.relpath(outdir, B)))

    if meta_all:
        _md = [m['md_chars'] for m in meta_all]
        _body = [m['chars']['total'] for m in meta_all]

        print('  首檔 %s（%s，%d 單位，md %s 字）'
              % (meta_all[0]['key'], meta_all[0]['vol'], meta_all[0]['n_units'],

    return stats

    r'^\s*(?:[-*•]|\d+[.、．])?\s*(?:\*\*|__|`)?\s*\[\s*(\d{1,3})\s*\]')
_CONT_RE = re.compile(r'^(?:\s{2,}|\t|(?:問題|依據|修正方向|→|說明)[：:])')
_SEC_RE = re.compile(r'(?m)^[ \t#*>\-]*【([^】\n]{1,24})】')


def _resolve(p):
    p = p.replace('/', os.sep)
    return p if os.path.isabs(p) else os.path.join(B, p)


def _rel(p):
    try:
        return os.path.relpath(p, B)
    except ValueError:
        return p


def _block(text, name):
    """取【name】區塊；標題可帶 markdown 前綴，區塊止於下一個【…】標題。"""
    marks = [(m.start(), m.end(), m.group(1).strip())
             for m in _SEC_RE.finditer(text or '')]
    for i, (s, e, title) in enumerate(marks):
        if title == name or title.startswith(name):
            end = marks[i + 1][0] if i + 1 < len(marks) else len(text)
            return text[e:end].strip()
    i = (text or '').find('【%s】' % name)
    if i < 0:
        return None
    j = text.find('\n【', i + len(name) + 2)
    return text[i + len(name) + 2:(j if j > 0 else len(text))].strip()


def _nums(s):
    out = set()
    s = re.sub(r'\]\s*[-–—~～至]\s*\[', '-', s or '')
    for m in _NUM_RE.finditer(s):
        a = int(m.group(1))
        b = int(m.group(2)) if m.group(2) else a
        if b < a:
            a, b = b, a
        if b - a > 500:
            out.add(a)
            continue
        out.update(range(a, b + 1))
    return out


def _is_none(txt):
    s = re.sub(r'[\s*_`#>\-•]', '', txt or '')
    return s in ('', '無', '無。', '无', '无。', '沒有', '没有')


def _parse_items(body, n_units):
    """回傳 (items, bad_numbers, unattributed)；bad_numbers 為越界或非數字段號。"""
    items, bad, loose = [], [], 0
    cur = None
    for raw in (body or '').splitlines():
        ln = raw.rstrip()
        if not ln.strip():
            continue
        m = _ITEM_RE.match(ln)
        if m:
            n = int(m.group(1))
            cur = {'n': n, 'text': ln.strip()}
            if 1 <= n <= n_units:
                items.append(cur)
            else:
                bad.append(n)
            continue
        if cur is not None and (ln.startswith(('  ', '\t')) or _CONT_RE.match(ln)
                                or not ln.lstrip().startswith('【')):
            cur['text'] += '\n' + ln.strip()
            continue
        cur = None
        if re.search(r'[須應補刪改修缺漏誤錯]', ln):
            loose += 1
    return items, bad, loose


def _unbalanced(t):
    for a, b, label in (('「', '」', '「」'), ('『', '』', '『』'),
                        ('（', '）', '（）'), ('《', '》', '《》'),
                        ('〔', '〕', '〔〕')):
        if t.count(a) != t.count(b):
            return label + ' 未成對'
    return ''


def _classify(reply_path, meta, key):
    """回傳 (status, reason, reply_text, items)。status: ok／pending／abnormal。"""
    rep_dir = os.path.dirname(reply_path)
    p400 = os.path.join(rep_dir, key + '_400審查.txt')
    pcut = os.path.join(rep_dir, key + '_截斷.txt')
    pabn = os.path.join(rep_dir, key + '_異常.txt')
    n = len(meta.get('units') or [])
    if os.path.exists(p400):
        return 'abnormal', '內容審查未通過（400）', None, []
    if not os.path.exists(reply_path):
        if os.path.exists(pcut):
            return 'abnormal', '截斷（僅有 _截斷.txt）', None, []
        if os.path.exists(pabn):
            return 'abnormal', '先前異常，尚未重跑（見 _異常.txt）', None, []
        return 'pending', '', None, []
    try:
        rt = io.open(reply_path, encoding='utf-8').read()
    except Exception as e:
        return 'abnormal', '回覆無法讀取：%s' % e, None, []
    if len(re.sub(r'\s', '', rt)) < 8:
        return 'abnormal', '回覆空白', rt, []
    tbl = _block(rt, '分段核驗表')
    if tbl is None:
        return 'abnormal', '缺【分段核驗表】', rt, []
    covered = _nums(tbl)
    if not covered:
        return 'abnormal', '【分段核驗表】無編號', rt, []
    missing = [i for i in range(1, n + 1) if i not in covered]
    extra = sorted(x for x in covered if not 1 <= x <= n)
    if missing:
        return ('abnormal', '漏號：%s' % '、'.join('[%02d]' % i for i in missing[:20]),
                rt, [])
    if extra:
        return ('abnormal', '編號越界：%s' % '、'.join('[%d]' % i for i in extra[:20]),
                rt, [])
    fixes = _block(rt, '必改硬傷')
    if fixes is None:
        return 'abnormal', '缺【必改硬傷】', rt, []
    if _is_none(fixes):
        return 'ok', '', rt, []
    if _unbalanced(rt):
        return 'abnormal', '疑似截斷（%s）' % _unbalanced(rt), rt, []
    items, bad, loose = _parse_items(fixes, n)
    if bad:
        return ('abnormal',
                '硬傷段號越界：%s' % '、'.join('[%d]' % x for x in bad[:20]), rt, [])
    if loose:
        return ('abnormal',
                '【必改硬傷】有 %d 行未標單位號（須以 [n] 起頭）' % loose, rt, [])
    if not items:
        return 'abnormal', '【必改硬傷】有內容但無法拆回', rt, []
    return 'ok', '', rt, items


def _write_abnormal(reply_path, key, reason, reply_text, dry_run):
    abn = os.path.join(os.path.dirname(reply_path), key + '_異常.txt')
    if dry_run:
        return abn
    body = '# 異常：%s\n\n' % reason
    if reply_text:
        body += reply_text
    else:
        body += '（本檔無可解析回覆；請看 %s_400審查.txt／%s_截斷.txt 或材料重跑）\n' % (key, key)
    io.open(abn, 'w', encoding='utf-8').write(body)
    if os.path.exists(reply_path):
        try:
            os.remove(reply_path)
        except OSError:
            pass
    return abn


def _unit_from_json(d):

def _bisect(meta, outdir, dry_run):
    """把一個異常 pack 對半切成兩份材料；回傳子 key 清單。"""
    import mk_reaudit_materials as RA
    units = [_unit_from_json(x) for x in (meta.get('units') or [])]
    n = len(units)
    if n < 2:
        return []
    mid = n // 2
    halves = [(units[:mid], '1/2'), (units[mid:], '2/2')]
    keys = []
    gaps_all = meta.get('gaps') or []
    for hi, (half, part) in enumerate(halves, 1):
        if not half:
            continue
        child = '%s_b%d' % (meta['key'], hi)
        if hi == 1:
            pre = (meta.get('pre_src', ''), meta.get('pre_tr', ''))
        else:
            pre = RA.context_before(units, mid)
        if hi == 2:
            post = (meta.get('post_src', ''), meta.get('post_tr', ''))
        else:
            post = RA.context_after(units, mid - 1)
        _ks = {u['key'] for u in half}
        gaps = [g for g in gaps_all
                if g.get('after') in _ks or g.get('before') in _ks]
        if not dry_run:
            RA.emit_pack(outdir, half, 0, meta.get('book', ''),
                         meta.get('book_id', ''), meta.get('count_mode', 'both'),
                         gaps, None, pre=pre, post=post,
                         parent=meta['key'], part=part, key=child)
        keys.append(child)
    return keys

def _term_table(unit, prev_u, next_u, cap=TERM_CAP):
    """本單位現譯與前／後鄰現譯共用、且多見於原文的專詞（2–4 字）。"""
    tr = _unit_tr(unit)

def _sec_bounds(text, name):

    return '\n'.join(L)

def _split_labeled(rt):
    """把「【疏文】…【記文】…」切成 [(lab, body)]；無標籤回 []。"""
    rt = FENCE_RE.sub('', rt or '')
    ms = list(_LAB_RE.finditer(rt))
    if not ms:
        return []
    out = []
    for i, m in enumerate(ms):
        end = ms[i + 1].start() if i + 1 < len(ms) else len(rt)
        body = rt[m.end():end]
        for cut in ('\n【必改硬傷】', '\n【可斟酌微瑕】', '\n【備註】', '\n【附註】'):
            if cut in body:
                body = body.split(cut)[0]
        out.append((m.group(1), _NUM_LEAD.sub('', body.strip()).strip()))
    return out


def _parse_reply(rt, parts):
    """回傳 (list[str] 對應 parts) 或 (None, 原因)。"""
    blocks = _split_labeled(rt)
    labs = [p.get('lab', '') for p in parts]
    if not blocks:
        body = FENCE_RE.sub('', rt or '').strip()
        for cut in ('【必改硬傷】', '【可斟酌微瑕】', '【分段核驗表】'):
            if cut in body:
                body = body.split(cut)[0]
        body = _NUM_LEAD.sub('', body.strip()).strip()
        if len(parts) == 1 and body:
            return [body], ''
        if len(parts) > 1:
            return None, '回覆無段別標籤（需 %s）' % '／'.join('【%s】' % x for x in labs)
        return None, '回覆為空'
    if len(blocks) != len(parts):
        return None, '段數不符（回覆 %d 段、定稿 %d 段）' % (len(blocks), len(parts))
    if len(parts) == 1 and len(blocks) == 1:

        # 標籤不作為退回理由，只取正文；多欄仍嚴格比對順序。
        return [blocks[0][1].strip()], ''

    if re.search(r'(?:local_search|web_search|web_fetch)\s*\(', out):
        issues.append('殘留工具呼叫字面')
    if re.search(r'(?m)^\s*(?:以下為.{0,16}整合|以下為整合|最終定稿白話版|完整修正版)', out):
        issues.append('殘留整合說明')
    if re.search(r'《(?:四分|十誦|五分|僧祇|摩訶僧祇)》', out):
        issues.append('律名簡稱未補「律」字')
    _labs = re.findall(r'(?m)^【(疏文|記文|鈔文|律文|註文|序文)】', out)
    if any(_labs.count(x) > 1 for x in set(_labs)):
        issues.append('殘留重複欄位標籤')
    for _q in re.findall(r'[「『]([^」』]{8,})[」』]', out):
        if (len(QC.HAN.findall(_q)) >= 10
                and re.search(r'[的了著嗎呢吧]', _q)
                and re.search(r'[乎者也矣焉哉]', _q)):
            issues.append('引號內疑半文半白')
            break
    _out_noquote = re.sub(r'[「『][^」』]*[」』]', '', out)
    for _bad, _good in QC.BAD_VAR.items():
        if _bad in _out_noquote:
            issues.append('異體字 %s' % _bad)
            break
    _simp = re.sub(r'[「『][^」』]*[」』]', '', out)
    if any(_b in _simp for _b in SIMP_BAD):
        issues.append('殘留簡體字')
    sh = len(QC.HAN.findall(src))
    oh = len(QC.HAN.findall(out))
    if sh >= 10 and oh < sh * 0.4:
        issues.append('嚴重縮水（疑漏譯）%d→%d' % (sh, oh))
    if 10 <= sh <= 75 and oh > sh * 2.4:
        issues.append('短文膨脹 %d→%d' % (sh, oh))
    _src_notes = max(0, src.count('（') + src.count('(') + src.count('〔') - out.count('〔'))
    _out_pars = [x.strip() for x in re.findall(r'[（(]([^）)]*)[）)]', out)
                 if not re.fullmatch(r'[一二三四五六七八九十百千\d]+', x.strip())
                 and not re.fullmatch(r'[^，。；]{0,8}反', x.strip())]
    if len(_out_pars) > _src_notes:
        issues.append('疑自加解說括號（譯 %d＞原 %d）' % (len(_out_pars), _src_notes))
    return issues


def _parts_of(row):
    segs = [s for s in (row.get('segments') or []) if (s.get('source') or s.get('translation'))]
    if segs:
        return [{'lab': s.get('label') or row.get('label') or '鈔文',
                 'orig': s.get('source') or '',
                 'tr': s.get('translation') or ''} for s in segs]
    return [{'lab': row.get('label') or '鈔文', 'orig': row.get('source') or '',
             'tr': row.get('translation') or ''}]


def _part_spans(row, raw_h):
    """回傳 parts 的 raw 起迄（累加漢字長；與 mk_reaudit_materials 同法）。"""
    a, b = int(row.get('raw_start', -1) or -1), int(row.get('raw_end', -1) or -1)
    parts = _parts_of(row)
    hs = [_han(p['orig']) for p in parts]
    if raw_h is None:
        return [(None, None) for _ in parts]
    unit_h = _han(row.get('source', ''))
    if 0 <= a < b <= len(raw_h) and raw_h[a:b] == unit_h \
            and sum(len(x) for x in hs) == len(unit_h):
        out, cur = [], a
        for h in hs:
            out.append((cur, cur + len(h)))
            cur += len(h)
        return out
    out, cur = [], max(0, a)
    for h in hs:
        q = raw_h.find(h, max(0, a - 40), b if b > a else len(raw_h)) if h else -1
        if q >= 0:
            out.append((q, q + len(h)))
            cur = q + len(h)
        else:
            out.append((cur, cur + len(h)))
            cur += len(h)
    return out


def _rebuild(old_tr, parts, new_trs):
    labs = [p['lab'] for p in parts]
    if len(new_trs) > 1:
        has_lab = any(('【%s】' % lab) in (old_tr or '') for lab in labs)
        if has_lab:
            return '\n'.join('【%s】%s' % (lab, tr) for lab, tr in zip(labs, new_trs))
        return '\n\n'.join(new_trs)
    tr = new_trs[0]
    prefix = re.match(r'^\s*(【[^】]+】)\s*', old_tr or '')
    if prefix and prefix.group(1) in ('【%s】' % x for x in LABS):
        sep = '\n' if re.match(r'^\s*【[^】]+】\s*\n', old_tr or '') else ''
        return '%s%s%s' % (prefix.group(1), sep, tr)
    return tr


def _guard_out(out, book):
    od = os.path.abspath(out)
    base = os.path.abspath(B)
    for bad in ('定稿', 'docs'):
        b = os.path.join(base, bad)
        if od == b or od.startswith(b + os.sep):
            raise SystemExit('拒絕寫入 %s（本流程不得改 定稿／docs）' % bad)
    if od == base:
        raise SystemExit('拒絕寫入專案根目錄')
    return od

def main():
    ap = argparse.ArgumentParser(
        description='再審套用：只替換 FIX 單位，其餘複製原定稿，產 定稿_v2（不呼叫 qwen）')
    ap.add_argument('--book', required=True, choices=list(BOOKS))
    ap.add_argument('--materials', default='',
                    help='再審材料目錄（預設 測試/重譯/<book>/_再審）')
    ap.add_argument('--plan', default='', help='拆回計畫（預設 <materials>/%s）' % PLAN_NAME)
    ap.add_argument('--revise', default='',
                    help='單段修訂材料目錄（預設 測試/重譯/<book>/_再審修訂）')
    ap.add_argument('--out', default='定稿_v2', help='輸出目錄（預設 定稿_v2）')
    ap.add_argument('--out-name', default='', help='輸出檔名（預設 <book_id>.jsonl）')
    ap.add_argument('--report', default='', help='對照報告路徑（預設 <out>/_再審變更對照.md）')
    ap.add_argument('--max-ratio', type=float, default=1.8, help='幅度上限（預設 1.8）')
    ap.add_argument('--min-ratio', type=float, default=0.55, help='幅度下限（預設 0.55）')
    ap.add_argument('--strict-qc', action='store_true', help='QC 有警示即退回不套用')
    ap.add_argument('--dry-run', action='store_true', help='只驗證與統計，不寫任何檔案')
    ap.add_argument('--force', action='store_true', help='允許覆寫既有輸出檔')
    args = ap.parse_args()

    bid = BOOKS[args.book]['id']
    out = _guard_out(_resolve(args.out), args.book)
    src_jsonl = os.path.join(B, '定稿', bid + '.jsonl')
    if not os.path.exists(src_jsonl):
        raise SystemExit('找不到定稿：' + src_jsonl)
    if not args.materials:
        args.materials = '測試/重譯/%s/_再審' % args.book
    if not args.revise:
        args.revise = '測試/重譯/%s/_再審修訂' % args.book
    mdir = _resolve(args.materials)
    plan_path = (_resolve(args.plan) if args.plan
                 else os.path.join(mdir, PLAN_NAME))
    if not os.path.exists(plan_path):
        raise SystemExit('找不到拆回計畫：' + plan_path)
    plan = json.load(io.open(plan_path, encoding='utf-8'))
    fix = plan.get('fix') or {}
    rep = os.path.join(_resolve(args.revise), '回覆')
    out_name = args.out_name or (bid + '.jsonl')
    dst = os.path.join(out, out_name)
    report = (_resolve(args.report) if args.report
              else os.path.join(out, '_再審變更對照.md'))
    if os.path.exists(dst) and not args.force and not args.dry_run:
        raise SystemExit('輸出檔已存在（%s）；請加 --force 或改用新目錄。' % dst)

    raw_h, raw_path = _load_raw(args.book)
    rows = [json.loads(x) for x in io.open(src_jsonl, encoding='utf-8') if x.strip()]
    by_key = {r.get('key'): r for r in rows}
    missing = [k for k in fix if k not in by_key]
    applied, blocked, pending = [], [], []
    new_rows = []
    for row in rows:
        key = row.get('key')
        if key not in fix:
            new_rows.append(row)
            continue
        info = fix[key]
        rp = os.path.join(rep, key + '_回.txt')
        if not os.path.exists(rp):
            pending.append((key, '缺修訂回覆'))
            new_rows.append(row)
            continue
        rt = io.open(rp, encoding='utf-8').read()
        parts = _parts_of(row)
        _segs = [s for s in (row.get('segments') or [])
                 if (s.get('source') or s.get('translation'))]
        if len(_segs) != len(parts):
            blocked.append((key, '段別結構不一致（segments %d、可譯 parts %d）'
                            % (len(_segs), len(parts))))
            new_rows.append(row)
            continue
        _why = _stale_reason(_unit_meta(mdir, info.get('pack'), key), row, parts)
        if _why:
            blocked.append((key, _why))
            new_rows.append(row)
            continue
        spans = _part_spans(row, raw_h)
        # source == raw 驗證
        raw_note = ''
        if raw_h is None:
            raw_note = 'raw 未驗（無底本）'
        else:
            a, b = int(row.get('raw_start', -1) or -1), int(row.get('raw_end', -1) or -1)
            if not (0 <= a < b <= len(raw_h) and raw_h[a:b] == _han(row.get('source', ''))):
                blocked.append((key, 'source≠raw（定稿定位與底本不符）'))
                new_rows.append(row)
                continue
            for p, (pa, pb) in zip(parts, spans):
                if pa is None or _han(p['orig']) != raw_h[pa:pb]:
                    raw_note = 'part 定位僅估算'
                    break
        new_trs, why = _parse_reply(rt, parts)
        if new_trs is None:
            blocked.append((key, why))
            new_rows.append(row)
            continue
        new_trs = [NR.normalize(t, p['orig']).strip() for t, p in zip(new_trs, parts)]
        if any(not _han(t) for t in new_trs):
            blocked.append((key, '回覆段別有空白譯文'))
            new_rows.append(row)
            continue
        flags = []
        old_han = sum(len(_han(p['tr'])) for p in parts)
        new_han = sum(len(_han(t)) for t in new_trs)
        ratio = (new_han / old_han) if old_han else 0.0
        if ratio and (ratio > args.max_ratio or ratio < args.min_ratio):
            flags.append('幅度 %.2f（%d→%d）' % (ratio, old_han, new_han))
        for p, t in zip(parts, new_trs):
            flags += _qc(p['orig'], t)
        if raw_note:
            flags.append(raw_note)
        if flags and args.strict_qc:
            blocked.append((key, '；'.join(flags)))
            new_rows.append(row)
            continue
        nr = copy.deepcopy(row)
        nr['translation'] = _rebuild(row.get('translation', ''), parts, new_trs)
        segs = nr.get('segments') or []
        for s, t in zip(segs, new_trs):
            s['translation'] = t
        if not segs and parts:
            nr['label'] = parts[0]['lab']
        nr['revised'] = True
        new_rows.append(nr)
        applied.append({
            'key': key, 'vol': row.get('vol', ''), 'chunk': row.get('chunk', ''),
            'labs': [p['lab'] for p in parts],
            'old': old_han, 'new': new_han, 'ratio': ratio,
            'flags': flags,
            'n_fix': len(info.get('items') or []),
        })

    # 驗證輸出結構
    if len(new_rows) != len(rows):
        raise SystemExit('內部錯誤：輸出筆數 %d ≠ 原定稿 %d' % (len(new_rows), len(rows)))
    keys_out = [r.get('key') for r in new_rows]
    if keys_out != [r.get('key') for r in rows]:
        raise SystemExit('內部錯誤：單位順序改變')

    print('再審套用（%s）：定稿 %d 單位；FIX %d；套用 %d、退回 %d、待修 %d'
          % (args.book, len(rows), len(fix), len(applied), len(blocked), len(pending)))
    for row in applied[:10]:
        print('  ✅ %s %s %s %d→%d 字%s'
              % (row['key'], row['vol'], '／'.join(row['labs']), row['old'],
                 row['new'], ('；' + '；'.join(row['flags'])) if row['flags'] else ''))
    for k, why in blocked[:10]:
        print('  ⛔ %s：%s' % (k, why))
    if args.dry_run:
        print('  （--dry-run：未寫入 %s）' % _rel(dst))
        return

    os.makedirs(out, exist_ok=True)
    with io.open(dst, 'w', encoding='utf-8') as f:
        for r in new_rows:
            f.write(json.dumps(r, ensure_ascii=False) + '\n')

    # ---- 對照報告 ----
    L = ['# 再審套用變更對照（%s）' % args.book, '',
         '- 來源定稿：%s（%d 單位）' % (_rel(src_jsonl), len(rows)),
         '- 輸出定稿：%s（%d 單位）' % (_rel(dst), len(new_rows)),
         '- 拆回計畫：%s' % _rel(plan_path),
         '- 原文底本：%s' % (raw_path or '（無，raw 驗證 unavailable）'),
         '- 套用 %d／退回 %d／待修 %d（FIX 共 %d）'
         % (len(applied), len(blocked), len(pending), len(fix)), '',
         '## 套用明細', '',
         '| # | key | 卷 | 段別 | 原譯字數 | 新譯字數 | 幅度 | 硬傷項 | QC／備註 |',
         '|--:|:--|:--|:--|--:|--:|--:|--:|:--|']
    for i, a in enumerate(applied, 1):
        L.append('| %d | %s | %s | %s | %d | %d | %.2f | %d | %s |'
                 % (i, a['key'], a['vol'], '／'.join(a['labs']), a['old'], a['new'],
                    a['ratio'], a['n_fix'], '；'.join(a['flags']) or '—'))
    L += ['', '## 未套用（退回）明細', '',
          '| key | 原因 |', '|:--|:--|']
    for k, why in blocked:
        L.append('| %s | %s |' % (k, why))
    if pending:
        L += ['', '## 待修（缺修訂回覆，保留原譯）', '',
              '| key | 原因 |', '|:--|:--|']
        for k, why in pending:
            L.append('| %s | %s |' % (k, why))
    if missing:
        L += ['', '## 計畫內有、定稿查無', '',
              '| key |', '|:--|']
        for k in missing:
            L.append('| %s |' % k)
    L += ['', '## 說明', '',
          '- 本檔只在 `--out` 寫出，未修改 `定稿/`、`docs/` 或任何網站檔。',
          '- 未列 FIX 的單位逐筆原樣複製（欄位、順序不變）。',
          '- 幅度＝新譯漢字／原譯漢字；超出 ±ratio 僅警示，`--strict-qc` 才退回。',
          '- QC 為 `qc_scan.py` 規則的同構掃描（該檔僅有 CLI，無可 import 的逐段函式）。']
    _rd = os.path.dirname(report)
    if _rd:
        os.makedirs(_rd, exist_ok=True)
    io.open(report, 'w', encoding='utf-8').write('\n'.join(L) + '\n')
    print('  輸出 → %s' % _rel(dst))
    print('  對照 → %s' % _rel(report))


if __name__ == '__main__':
    main()
