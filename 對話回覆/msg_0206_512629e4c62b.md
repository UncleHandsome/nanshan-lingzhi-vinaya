def _strip_dup_labels(out):
    seen = set()
    lines = out.split('\n')
    res = []
    for ln in lines:
        m = re.match(r'^(【(?:疏文|記文|鈔文)】)\s*', ln)
        if m:
            lab = m.group(1)
            if lab in seen:
                ln = ln[m.end():]      # 去掉重複標籤
            else:
                seen.add(lab)
        res.append(ln)
    return '\n'.join(res)
```

def _looks_pasted(out, src_text):
    HAN = re.compile(r'[\u4e00-\u9fff]')
    a = ''.join(HAN.findall(out))[:20]
    b = ''.join(HAN.findall(src_text))[:20]
    if len(a) < 8 or len(b) < 8: return False
    # 共同前綴比例
    n = 0
    for x, y in zip(a, b):
        if x == y: n += 1
        else: break
    return n >= 12
```
