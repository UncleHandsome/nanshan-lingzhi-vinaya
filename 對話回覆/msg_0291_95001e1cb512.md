## 四、合稿後零成本自動品檢（建議新增 `_src/qc_check.py`）

接在 `build_a2.py` 之後跑，不花任何 token，自動把 `L4` 那類偶發漏檢（引號改白話、引號外漏譯、目錄暴長、殘留網址／異體字）全部掃成人工覆核清單：

```python
# -*- coding: utf-8 -*-
"""定稿自動體檢：零 LLM 成本抓出引號改寫、短段膨脹與殘跡。"""
import argparse, io, os, re

B = r'C:\Users\lionc\OneDrive\桌面\XW\三大部'
HAN = re.compile(r'[\u4e00-\u9fff]')
QUO = re.compile(r'「([^」]{6,})」')
DE_RE = re.compile(r'[的了著嗎呢吧啊]')

def hanzis(s):
    return ''.join(HAN.findall(s))

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--base', required=True, help='裁決目錄（讀原文用）')
    ap.add_argument('--final', required=True, help='定稿目錄（.txt）')
    args = ap.parse_args()

    base = os.path.join(B, args.base.replace('/', os.sep))
    fin = os.path.join(B, args.final.replace('/', os.sep))

    for fn in sorted(os.listdir(fin)):
        if not fn.endswith('.txt'):
            continue
        key = fn[:-4]
        out = io.open(os.path.join(fin, fn), encoding='utf-8').read()
        src_p = os.path.join(base, key + '.md')
        src = io.open(src_p, encoding='utf-8').read() if os.path.exists(src_p) else ''
        orig = ''.join(re.findall(r'■ (?:疏文|記文|鈔文|律文|註文|序文)（原文）\s*\n+(.+?)(?=\n+■ |\n+【後文背景|\Z)', src, re.S))
        orig_h = hanzis(orig)
        warns = []

        # 1. 引號內疑改寫為白話（長引號內漢字未見於原文且含白話虛詞）
        for q in QUO.findall(out):
            qh = hanzis(q)
            if len(qh) >= 8 and qh not in orig_h and DE_RE.search(q):
                warns.append('引號疑改白話:「%s」' % q[:18])

        # 2. 極短段異常膨脹（如目錄擅自編造解釋）
        if 10 <= len(orig_h) <= 75 and len(hanzis(out)) > len(orig_h) * 2.4:
            warns.append('短段膨脹(原%d→譯%d字)' % (len(orig_h), len(hanzis(out))))

        # 3. 網址、殘留檢索或常見異體字（僅供檢查）
        if re.search(r'https?://|見[^。\n]{0,12}辭典', out):
            warns.append('疑似殘留網址/出處')
        if '鉢' in out and '鉢' not in orig:
            warns.append('含非標準異體字「鉢」')

        if warns:
            print(f'[{key}] ' + ' ｜ '.join(warns))

if __name__ == '__main__':
    main()
```
